### **Basic Array Operations**

#### **Solution 1: Array Initialization and Traversal**
```java
public class Main {
    public static void main(String[] args) {
        int[] arr = {10, 20, 30, 40, 50};
        for (int value : arr) {
            System.out.print(value + " ");
        }
    }
}
```

#### **Solution 2: Copy an Array**
```java
public class Main {
    public static void main(String[] args) {
        int[] original = {1, 2, 3, 4, 5};
        int[] copy = new int[original.length];

        for (int i = 0; i < original.length; i++) {
            copy[i] = original[i];
        }

        for (int value : copy) {
            System.out.print(value + " ");
        }
    }
}
```

#### **Solution 3: Array Search**
```java
public class Main {
    public static void main(String[] args) {
        int[] arr = {10, 15, 20, 25, 30};
        int target = 20;
        boolean found = false;

        for (int i = 0; i < arr.length; i++) {
            if (arr[i] == target) {
                System.out.println("Element found at index: " + i);
                found = true;
                break;
            }
        }

        if (!found) {
            System.out.println("Element not found.");
        }
    }
}
```

#### **Solution 4: Remove an Element**

```java
public class Main {
    public static void main(String[] args) {
        int[] arr = {10, 20, 30, 40, 50};
        int indexToRemove = 2;

        for (int i = indexToRemove; i < arr.length - 1; i++) {
            arr[i] = arr[i + 1];
        }

        System.out.print("Array after removal: ");
        for (int i = 0; i < arr.length - 1; i++) {
            System.out.print(arr[i] + " ");
        }
    }
```

#### **Solution 5: Insert an Element**
```java
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 2, 3, 4, 5};
        int[] newArr = new int[arr.length + 1];
        int newElement = 10;
        int indexToInsert = 2;

        for (int i = 0; i < indexToInsert; i++) {
            newArr[i] = arr[i];
        }

        newArr[indexToInsert] = newElement;

        for (int i = indexToInsert; i < arr.length; i++) {
            newArr[i + 1] = arr[i];
        }

        System.out.println("Array after insertion: " + Arrays.toString(newArr));
    }
}
```

#### **Solution 6: Merge Two Arrays**
```java
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        int[] arr1 = {1, 2, 3};
        int[] arr2 = {4, 5, 6};
        int[] merged = new int[arr1.length + arr2.length];

        for (int i = 0; i < arr1.length; i++) {
            merged[i] = arr1[i];
        }

        for (int i = 0; i < arr2.length; i++) {
            merged[arr1.length + i] = arr2[i];
        }

        System.out.println("Merged array: " + Arrays.toString(merged));
    }
}
```

#### **Solution 11: Count Occurrences**
```java
import java.util.HashMap;

public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 2, 2, 3, 4, 4, 4, 5};
        HashMap<Integer, Integer> counts = new HashMap<>();

        for (int value : arr) {
            counts.put(value, counts.getOrDefault(value, 0) + 1);
        }

        for (int key : counts.keySet()) {
            System.out.println(key + " occurs " + counts.get(key) + " times.");
        }
    }
}
```

#### **Solution 12: Rotate an Array**
```java
public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 2, 3, 4, 5};
        int rotations = 2;

        int[] rotated = new int[arr.length];
        for (int i = 0; i < arr.length; i++) {
            rotated[(i + rotations) % arr.length] = arr[i];
        }

        System.out.print("Rotated array: ");
        for (int value : rotated) {
            System.out.print(value + " ");
        }
    }
}
```

#### **Solution 13: Check Palindrome Array**

```java
public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 2, 3, 2, 1};
        boolean isPalindrome = true;

        for (int i = 0; i < arr.length / 2; i++) {
            if (arr[i] != arr[arr.length - 1 - i]) {
                isPalindrome = false;
                break;
            }
        }

        System.out.println("Is palindrome: " + isPalindrome);
    }
}
```

#### **Solution 14: Find Missing Number**

```java
public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 2, 3, 4, 5, 7, 8, 9, 10};
        int n = 10;
        int sumExpected = n * (n + 1) / 2;
        int sumActual = 0;

        for (int value : arr) {
            sumActual += value;
        }

        System.out.println("Missing number: " + (sumExpected - sumActual));
    }
}
```
