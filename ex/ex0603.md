#### **Solution 11: Count Vowels and Consonants**
```java
public class Main {
    public static void main(String[] args) {
        String str = "Java Programming".toLowerCase();
        int vowels = 0, consonants = 0;

        for (char ch : str.toCharArray()) {
            if (ch >= 'a' && ch <= 'z') {
                if ("aeiou".indexOf(ch) != -1) {
                    vowels++;
                } else {
                    consonants++;
                }
            }
        }

        System.out.println("Vowels: " + vowels);
        System.out.println("Consonants: " + consonants);
    }
}
```

#### **Solution 15: String Compression**
```java
public class Main {
    public static void main(String[] args) {
        String str = "aaabbcc";
        StringBuilder compressed = new StringBuilder();

        int count = 1;
        for (int i = 1; i < str.length(); i++) {
            if (str.charAt(i) == str.charAt(i - 1)) {
                count++;
            } else {
                compressed.append(str.charAt(i - 1)).append(count);
                count = 1;
            }
        }
        compressed.append(str.charAt(str.length() - 1)).append(count);

        System.out.println("Compressed String: " + compressed);
    }
}
```

### **StringBuilder vs StringBuffer**

#### **Solution 16: Performance Comparison**
```java
public class Main {
    public static void main(String[] args) {
        long startTime, endTime;

        StringBuilder sb = new StringBuilder();
        startTime = System.nanoTime();
        for (int i = 0; i < 100000; i++) {
            sb.append("a");
        }
        endTime = System.nanoTime();
        System.out.println("StringBuilder Time: " + (endTime - startTime) + " ns");

        StringBuffer sbf = new StringBuffer();
        startTime = System.nanoTime();
        for (int i = 0; i < 100000; i++) {
            sbf.append("a");
        }
        endTime = System.nanoTime();
        System.out.println("StringBuffer Time: " + (endTime - startTime) + " ns");
    }
}
```

#### **Solution 17: Thread-Safety**
```java
public class Main {
    public static void main(String[] args) {
        StringBuffer buffer = new StringBuffer();
        Runnable task = () -> {
            for (int i = 0; i < 10; i++) {
                buffer.append(Thread.currentThread().getName()).append(" ");
            }
        };

        Thread t1 = new Thread(task, "Thread1");
        Thread t2 = new Thread(task, "Thread2");

        t1.start();
        t2.start();

        try {
            t1.join();
            t2.join();
        } catch (InterruptedException e) {
            e.printStackTrace();
        }

        System.out.println(buffer.toString());
    }
}
```

#### **Solution 19: Word Frequency**
```java
import java.util.HashMap;

public class Main {
    public static void main(String[] args) {
        String str = "apple banana apple grape banana apple";
        String[] words = str.split(" ");

        HashMap<String, Integer> freq = new HashMap<>();
        for (String word : words) {
            freq.put(word, freq.getOrDefault(word, 0) + 1);
        }

        freq.forEach((word, count) -> System.out.println(word + ": " + count));
    }
}
```

#### **Solution 20: Remove Duplicates from String**
```java
import java.util.LinkedHashSet;

public class Main {
    public static void main(String[] args) {
        String str = "programming";
        LinkedHashSet<Character> set = new LinkedHashSet<>();

        for (char ch : str.toCharArray()) {
            set.add(ch);
        }

        StringBuilder result = new StringBuilder();
        for (char ch : set) {
            result.append(ch);
        }

        System.out.println("String without duplicates: " + result);
    }
}
```
