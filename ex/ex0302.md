# Answers

Here are the answers to the 15 loop-related programming questions in Java:

---

### **1. Print Numbers**
```java
public class PrintNumbers {
    public static void main(String[] args) {
        for (int i = 1; i <= 10; i++) {
            System.out.println(i);
        }
    }
}
```

---

### **2. Sum of Natural Numbers**
```java
public class SumOfNaturalNumbers {
    public static void main(String[] args) {
        int sum = 0, i = 1;
        while (i <= 10) {
            sum += i;
            i++;
        }
        System.out.println("Sum: " + sum);
    }
}
```

### **3. Factorial of a Number**

```java
public class Factorial {
    public static void main(String[] args) {
        int number = 5, factorial = 1;

}
```

### **8. Fibonacci Series**
```java
public class FibonacciSeries {
    public static void main(String[] args) {
        int n = 10, a = 0, b = 1;
        System.out.println("Fibonacci series:");
        for (int i = 1; i <= n; i++) {
            System.out.println(a);
            int next = a + b;
            a = b;
            b = next;
        }
    }
}
```

### **9. Check Prime**

```java
public class CheckPrime {

        int number = 29; // Example number
        boolean isPrime = true;
        for (int i = 2; i <= number / 2; i++) {
```

### **11. Reverse a Number**
```java
public class ReverseNumber {
    public static void main(String[] args) {
        int number = 1234, reverse = 0;
        while (number > 0) {
            int digit = number % 10;
            reverse = reverse * 10 + digit;
            number /= 10;
        }
        System.out.println("Reversed number: " + reverse);
    }
}
```

### **13. Palindrome Check**
```java
public class PalindromeCheck {
    public static void main(String[] args) {
        int number = 121, original = number, reverse = 0;
        while (number > 0) {
            int digit = number % 10;
            reverse = reverse * 10 + digit;
            number /= 10;
        }
        System.out.println(original + " is " + (original == reverse ? "a palindrome." : "not a palindrome."));
    }
}
```

### **14. Number Pyramid**
```java
public class NumberPyramid {
    public static void main(String[] args) {
        int n = 4;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int j = 1; j <= i; j++) {
                System.out.print(j);
            }
            for (int j = i - 1; j >= 1; j--) {
                System.out.print(j);
            }
            System.out.println();
        }
    }
}
```

---

### **15. Armstrong Number**
```java
public class ArmstrongNumber {
    public static void main(String[] args) {
        int number = 153, original = number, result = 0, digits = 0;
        while (number > 0) {
            digits++;
            number /= 10;
        }
        number = original;
        while (number > 0) {
            int digit = number % 10;
            result += Math.pow(digit, digits);
            number /= 10;
        }
        System.out.println(original + " is " + (original == result ? "an Armstrong number." : "not an Armstrong number."));
    }
}
```
